%{
#include <stdio.h>
int count = 0;
%}

%%
VU      { count++; }
.|\n    ;  // Ignore all other characters
%%

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Usage: %s <input_file>\n", argv[0]);
        return 1;
    }

    FILE *file = fopen(argv[1], "r");
    if (!file) {
        perror("Error opening file");
        return 1;
    }

    yyin = file;   // Redirect lex input from file
    yylex();       // Start scanning
    fclose(file);

    printf("The word \"VU\" appeared %d times.\n", count);
    return 0;
}

int yywrap() {
    return 1;
}
